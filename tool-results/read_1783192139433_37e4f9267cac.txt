   177→   177→   SUB-NAVBAR — Sticky scroll-spy navigation + DEBOUNCE anti-jitter
   178→   178→   ═══════════════════════════════════════════════════════════════════ */
   179→   179→
   180→   180→function SubNavbar({ categories }: { categories: typeof categories }) {
   181→   181→  const [activeId, setActiveId] = useState(categories[0].id)
   182→   182→  const navRef = useRef<HTMLDivElement>(null)
   183→   183→  const blockedRef = useRef(false)
   184→   184→
   185→   185→  // Scroll tab into view using ONLY horizontal scrollLeft (never touches page scroll)
   186→   186→  const scrollTabIntoView = useCallback((catId: string) => {
   187→   187→    const nav = navRef.current
   188→   188→    if (!nav) return
   189→   189→    const tab = nav.querySelector(`[data-cat="${catId}"]`) as HTMLElement | null
   190→   190→    if (!tab) return
   191→   191→    // Calculate if tab is outside visible area of nav
   192→   192→    const navRect = nav.getBoundingClientRect()
   193→   193→    const tabRect = tab.getBoundingClientRect()
   194→   194→    const tabCenter = tabRect.left - navRect.left + tabRect.width / 2
   195→   195→    const navCenter = navRect.width / 2
   196→   196→    const diff = tabCenter - navCenter
   197→   197→    if (Math.abs(diff) > 20) {
   198→   198→      nav.scrollBy({ left: diff, behavior: 'smooth' })
   199→   199→    }
   200→   200→  }, [])
   201→   201→
   202→   202→  useEffect(() => {
   203→   203→    const observers: IntersectionObserver[] = []
   204→   204→
   205→   205→    categories.forEach((cat) => {
   206→   206→      const el = document.getElementById(cat.id)
   207→   207→      if (!el) return
   208→   208→
   209→   209→      const observer = new IntersectionObserver(
   210→   210→        ([entry]) => {
   211→   211→          // If user clicked a tab, observer sleeps for 1 second
   212→   212→          if (blockedRef.current) return
   213→   213→
   214→   214→          if (entry.isIntersecting) {
   215→   215→            setActiveId(cat.id)
   216→   216→            scrollTabIntoView(cat.id)
   217→   217→          }
   218→   218→        },
   219→   219→        { rootMargin: '-40% 0px -40% 0px', threshold: 0 }
   220→   220→      )
   221→   221→      observer.observe(el)
   222→   222→      observers.push(observer)
   223→   223→    })
   224→   224→
   225→   225→    return () => observers.forEach((o) => o.disconnect())
   226→   226→  }, [categories, scrollTabIntoView])
   227→   227→
   228→   228→  const handleClick = useCallback((id: string) => {
   229→   229→    const el = document.getElementById(id)
   230→   230→    if (!el) return
   231→   231→
   232→   232→    // Immediately activate the clicked tab
   233→   233→    setActiveId(id)
   234→   234→    scrollTabIntoView(id)
   235→   235→
   236→   236→    // Block observer for 1 full second to prevent any scroll jitter
   237→   237→    blockedRef.current = true
   238→   238→    setTimeout(() => { blockedRef.current = false }, 1000)
   239→   239→
   240→   240→    // Smooth scroll to category with 150px offset
   241→   241→    const offset = 150
   242→   242→    const top = el.getBoundingClientRect().top + window.scrollY - offset
   243→   243→    window.scrollTo({ top, behavior: 'smooth' })
   244→   244→  }, [scrollTabIntoView])
   245→   245→
   246→   246→  return (
   247→   247→    <div className="sticky top-[56px] sm:top-[60px] lg:top-[64px] z-[35] bg-white/90 backdrop-blur-xl border-b border-marine/5 shadow-[0_2px_20px_rgba(0,32,96,0.04)]">
   248→   248→      <div className="w-full max-w-[1200px] mx-auto px-4 sm:px-6 lg:px-8">
   249→   249→        <nav
   250→   250→          ref={navRef}
   251→   251→          className="flex items-center gap-1 overflow-x-auto py-2.5 scrollbar-none -mx-4 px-4 sm:mx-0 sm:px-0"
   252→   252→          style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}
   253→   253→        >
   254→   254→          {categories.map((cat) => (
   255→   255→            <button
   256→   256→              key={cat.id}
   257→   257→              data-cat={cat.id}
   258→   258→              onClick={() => handleClick(cat.id)}
   259→   259→              className={`flex items-center gap-2 px-4 py-2 rounded-full text-sm font-medium whitespace-nowrap transition-all duration-300 flex-shrink-0 ${
   260→   260→                activeId === cat.id
   261→   261→                  ? 'bg-marine text-white shadow-md shadow-marine/20'
   262→   262→                  : 'text-marine/60 hover:text-marine hover:bg-marine/5'
   263→   263→              }`}
   264→   264→            >
   265→   265→              <cat.icon className="w-4 h-4" />
   266→   266→              <span className="hidden sm:inline">{cat.label}</span>
   267→   267→              <span className="sm:hidden">{cat.shortLabel}</span>
   268→   268→            </button>
   269→   269→          ))}
   270→   270→        </nav>
   271→   271→      </div>
   272→   272→    </div>
   273→   273→  )
   274→   274→}
